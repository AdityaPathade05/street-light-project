Report Street Light Issue

function submitIssue() { let lampId = document.getElementById("lampId").value; let issue = document.getElementById("issue").value; if (lampId === "") { alert("Please enter Lamp ID"); return; } let reports = JSON.parse(localStorage.getItem("reports")) || []; reports.push({ lampId: lampId, issue: issue, time: new Date().toLocaleString() }); localStorage.setItem("reports", JSON.stringify(reports)); document.getElementById("msg").innerText = "Issue reported successfully!"; }
